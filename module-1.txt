7. MONGODB SCHEMA
   Database: ai_interview_system
   
   Collections:
   
   a) users (Admin users)
   {
     "_id": ObjectId,
     "email": String (unique, indexed),
     "password_hash": String,
     "full_name": String,
     "role": String (default: "admin"),
     "created_at": DateTime
   }
   
   b) candidates
   {
     "_id": ObjectId,
     "name": String,
     "email": String,
     "phone": String,
     "resume_url": String,
     "skills": Array[String],
     "experience": String,
     "education": String,
     "projects": Array[{
       "title": String,
       "description": String,
       "technologies": Array[String]
     }],
     "achievements": Array[String],
     "summary": String,
     "parsed_data": Object,
     "created_at": DateTime,
     "created_by": ObjectId (ref: users)
   }
   
   c) interviews
   {
     "_id": ObjectId,
     "interview_id": String (unique, 8-char, indexed),
     "candidate_id": ObjectId (ref: candidates),
     "admin_id": ObjectId (ref: users),
     "job_description": String,
     "interview_type": String (enum: "ai_based", "ai_assisted"),
     "status": String (enum: "scheduled", "in_progress", "completed", "expired"),
     "scheduled_at": DateTime,
     "expiry_date": DateTime,
     "started_at": DateTime,
     "ended_at": DateTime,
     "created_at": DateTime
   }
   
   d) transcripts
   {
     "_id": ObjectId,
     "interview_id": String (ref: interviews.interview_id),
     "speaker": String (enum: "ai", "candidate", "interviewer"),
     "text": String,
     "timestamp": DateTime,
     "audio_url": String (optional),
     "sequence": Number
   }
   
   e) reports
   {
     "_id": ObjectId,
     "interview_id": String (ref: interviews.interview_id),
     "qa_mapping": Array[{
       "question": String,
       "answer": String,
       "evaluation": String,
       "score": Number
     }],
     "performance_score": Number (0-100),
     "strengths": Array[String],
     "weaknesses": Array[String],
     "technical_assessment": {
       "coding_skills": Number,
       "problem_solving": Number,
       "communication": Number,
       "technical_knowledge": Number
     },
     "recommendation": String (enum: "hire", "reject", "maybe"),
     "detailed_analysis": String,
     "generated_at": DateTime
   }

9. DATABASE CONNECTION
   File: backend/app/core/database.py
   ```python
   from motor.motor_asyncio import AsyncIOMotorClient
   import os
   from dotenv import load_dotenv
   
   load_dotenv()
   
   MONGODB_URL = os.getenv("MONGODB_URL")
   DATABASE_NAME = os.getenv("DATABASE_NAME")
   
   # Async client for FastAPI
   client = AsyncIOMotorClient(MONGODB_URL)
   database = client[DATABASE_NAME]
   
   # Collections
   users_collection = database.get_collection("users")
   candidates_collection = database.get_collection("candidates")
   interviews_collection = database.get_collection("interviews")
   transcripts_collection = database.get_collection("transcripts")
   reports_collection = database.get_collection("reports")
   
   # Create indexes
   async def create_indexes():
       await users_collection.create_index("email", unique=True)
       await interviews_collection.create_index("interview_id", unique=True)
       await transcripts_collection.create_index([("interview_id", 1), ("sequence", 1)])
   ```

10. BASIC FASTAPI APP
    File: backend/app/main.py
    ```python
    from fastapi import FastAPI
    from fastapi.middleware.cors import CORSMiddleware
    from dotenv import load_dotenv
    from .core.database import create_indexes
    import os
    
    load_dotenv()
    
    app = FastAPI(
        title="AI Interview System API",
        description="Backend API for AI-powered interview platform",
        version="1.0.0"
    )
    
    # CORS configuration
    app.add_middleware(
        CORSMiddleware,
        allow_origins=[
            os.getenv("ADMIN_FRONTEND_URL", "http://localhost:3000"),
            os.getenv("CANDIDATE_FRONTEND_URL", "http://localhost:3001")
        ],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )
    
    @app.on_event("startup")
    async def startup_event():
        await create_indexes()
    
    @app.get("/")
    async def root():
        return {
            "message": "AI Interview System API",
            "version": "1.0.0",
            "status": "running"
        }
    
    @app.get("/health")
    async def health_check():
        return {"status": "healthy"}
    ```
